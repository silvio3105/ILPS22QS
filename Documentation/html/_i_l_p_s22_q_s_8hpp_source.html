414 Return_t init(const InterfaceConfig_s* interfaceCfg = nullptr)
415 {
416 PRINTF_INFO("ILPS22QS ", 9);
417 PRINTF_INFO(version, sizeof(version) - 1);
418 PRINTF_INFO("\n", 1);
419
420 // Check interface object
421 if (interface.check() != Return_t::OK)
422 {
423 PRINTN_ERROR("ILPS22QS ", 9);
424 PRINTN_ERROR("Check fail\n", 11);
425 return Return_t::NOK;
426 }
427
428 // Call MSP init handler if provided
429 if (mspInitHandler)
430 {
431 if (mspInitHandler() != Return_t::OK)
432 {
433 PRINTN_ERROR("ILPS22QS ", 9);
434 PRINTN_ERROR("MSP init fail\n", 14);
435 return Return_t::NOK;
436 }
437 }
438
439 // Set interface config
440 if (interfaceCfg)
441 {
442 if (interfaceConfig(interfaceCfg) != Return_t::OK)
443 {
444 PRINTN_ERROR("ILPS22QS ", 9);
445 PRINTN_ERROR("Cfg fail\n", 9);
446 return Return_t::NOK;
447 }
448 }
449
450 // Check sensor ID
451 uint8_t tmp = 0;
452 if (whoAmI(tmp) == Return_t::OK)
453 {
454 if (tmp != chipID)
455 {
456 PRINTN_ERROR("ILPS22QS ", 9);
457 PRINTN_ERROR("Invalid ID\n", 11);
458 return Return_t::NOK;
459 }
460 }
461 else
462 {
463 PRINTN_ERROR("ILPS22QS ", 9);
464 PRINTN_ERROR("WhoAmI fail\n", 12);
465 return Return_t::NOK;
466 }
467
468 // Cache pressure scale
469 return readPressureScale();
470 }

549 Return_t setPressureInterruptThreshold(const uint16_t threshold)
550 {
551 // Calculate raw pressure value and set TX buffer
552 const uint16_t rawPressure = threshold * pressureScaleDivider[(uint8_t)getPressureScale()];
553 txBuffer[0] = (uint8_t)Register_t::PressureThresholdHigh;
554 txBuffer[1] = rawPressure >> 8;
555
556 if (writeRegister(txBuffer, 2) != Return_t::OK)
557 {
558 return Return_t::NOK;
559 }
560
561 txBuffer[0] = (uint8_t)Register_t::PressureThresholdLow;
562 txBuffer[1] = rawPressure & 0xFF;
563
564 if (writeRegister(txBuffer, 2) != Return_t::OK)
565 {
566 return Return_t::NOK;
567 }
568
569 PRINTN_INFO("ILPS22QS ", 9);
570 PRINTF_INFO("Interrupt threshold set to %u\n", threshold);
571 return Return_t::OK;
572 }

582 Return_t getPressureInterruptThreshold(uint16_t& output)
583 {
584 uint16_t tmpOutput = 0;
585 uint8_t tmp = 0;
586
587 // Read high byte
588 if (readRegister(Register_t::PressureThresholdHigh, tmp) != Return_t::OK)
589 {
590 return Return_t::NOK;
591 }
592 tmpOutput = tmp << 8;
593
594 // Read low byte
595 if (readRegister(Register_t::PressureThresholdLow, tmp) != Return_t::OK)
596 {
597 return Return_t::NOK;
598 }
599 tmpOutput |= tmp;
600
601 // Convert to output unit
602 output = tmpOutput / pressureScaleDivider[(uint8_t)getPressureScale()];
603 return Return_t::OK;
604 }

614 Return_t setPressureScale(const PressureScale_t scale)
615 {
616 // Read control 2 register
617 uint8_t tmp = 0;
618
619 if (readRegister(Register_t::Control2, tmp) != Return_t::OK)
620 {
621 return Return_t::NOK;
622 }
623
624 txBuffer[0] = (uint8_t)Register_t::Control2;
625 txBuffer[1] = tmp & ~(1 << (uint8_t)Control2Bitmap_t::FullScale);
626 txBuffer[1] |= ((uint8_t)scale << (uint8_t)Control2Bitmap_t::FullScale);
627
628 if (writeRegister(txBuffer, 2) != Return_t::OK)
629 {
630 return Return_t::NOK;
631 }
632
633 pressureScale = scale;
634
635 PRINTN_INFO("ILPS22QS ", 9);
636 PRINTF_INFO("Pressure scale set to %02X\n", txBuffer[1]);
637 return Return_t::OK;
638 }

1064 Return_t getPressure(uint16_t& output)
1065 {
1066 int32_t tmp = 0;
1067 uint8_t reg = 0;
1068
1069 if (readRegister(Register_t::PressureOutHigh, reg) != Return_t::OK)
1070 {
1071 return Return_t::NOK;
1072 }
1073 tmp = reg << 16;
1074
1075 if (readRegister(Register_t::PressureOutMid, reg) != Return_t::OK)
1076 {
1077 return Return_t::NOK;
1078 }
1079 tmp |= reg << 8;
1080
1081 if (readRegister(Register_t::PressureOutLow, reg) != Return_t::OK)
1082 {
1083 return Return_t::NOK;
1084 }
1085 tmp |= reg;
1086
1087 static constexpr uint16_t scaleDivider[] = { 4096, 2048 };
1088 output = tmp / scaleDivider[(uint8_t)getPressureScale()];
1089
1090 PRINTN("ILPS22QS ", 9);
